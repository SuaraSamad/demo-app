FIX 2 - App.jsx: turn loading on and off
Select the whole handleSave function and replace it with:

  async function handleSave() {
    setLoading(true)
    setMessage('')
    setRequests((requests) => requests + 1)
    await saveToServer()
    setMessage(`Saved "${name}"`)
    setLoading(false)
  }
